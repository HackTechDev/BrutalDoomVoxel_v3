	+NOBLOCKMAP
	+FLOORCLIP
	+FORCEXYBILLBOARD
	+NOINTERACTION
	+FORCEXYBILLBOARD
	+DONTSPLASH
	+INTERPOLATEANGLES
	+MISSILE
	Speed 1
	RenderStyle Add
	Alpha		0.06
	+CLIENTSIDEONLY
	Radius		0
	Height		0
	Scale		0.5
	States
	{
	Spawn:
		TNT1 A 1

		Stop
	}
